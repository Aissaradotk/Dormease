       FROM dorm_bills b
       JOIN dorm_rooms r ON b.room_id = r.id
       WHERE (b.tenant_id = ? OR (b.room_id = ? AND b.tenant_id IS NULL))
       ORDER BY b.period DESC, b.created_at DESC`,
      [tenant.id, tenant.room_id]
    );

    res.json(bills);
  } catch (err) {

  }
